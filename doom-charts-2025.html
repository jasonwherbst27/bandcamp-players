One-Shot Local Auto-Resolver (Python 3)
import json, re, urllib.request html = open("doom_charts_2025_archive_player.html", encoding="utf-8").read() m = re.search(r"const TRACKS = (\[.*?\]);", html, re.S) tracks = json.loads(m.group(1)) for t in tracks: if t.get("album_id") or "bandcamp.com/album/" not in (t.get("url") or ""): continue try: req = urllib.request.Request(t["url"], headers={"User-Agent": "Mozilla/5.0"}) page = urllib.request.urlopen(req, timeout=15).read().decode("utf-8", "ignore") hit = re.search(r"album=(\d+)", page) or re.search(r'"id":(\d+),"type":"album"', page) if hit: t["album_id"] = int(hit.group(1)) print(f"#{t['rank']:02d} {t['band']} -> {t['album_id']}") except Exception as e: print(f"#{t['rank']:02d} skipped: {e}") out = html[:m.start(1)] + json.dumps(tracks, ensure_ascii=False) + html[m.end(1):] open("doom_charts_2025_googie_embedded.html", "w", encoding="utf-8").write(out) print("Saved doom_charts_2025_googie_embedded.html")
